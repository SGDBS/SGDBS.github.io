#### 代码

```
#include <cstdio>
#include <iostream>
#include <algorithm>
using namespace std;

const int maxN = 1e6 + 7;
const long long INF = 0x7f7f7f7f;

int T, n, ans[maxN];
long long s[maxN][2];

struct Num {
    long long x, y;
    int id;
}a[maxN];

bool cmp(Num a, Num b)
{
    return a.x < b.x;
}

int main()
{
    scanf("%d", &T);
    while(T--) {
        scanf("%d", &n);
        for(int i = 1; i <= n; ++i)
            a[i].id = i;
        for(int i = 1; i <= n; ++i)
            scanf("%lld", &a[i].x);
        for(int i = 1; i <= n; ++i)
            scanf("%lld", &a[i].y);
        sort(a + 1, a + 1 + n, cmp);
        int pos = 0;
        s[n + 1][1] = INF;
        for(int i = 1; i <= n; ++i)
            s[i][0] = max(s[i - 1][0], a[i].y);
        for(int i = n; i >= 1; --i)
            s[i][1] = min(s[i + 1][1], a[i].y);
        for(int i = n; i >= 1; --i)
            if(s[i][1] > s[i - 1][0]) {
                pos = i;
                break;
            }
        for(int i = 1; i < pos; ++i)
            ans[a[i].id] = 0;
        for(int i = pos; i <= n; ++i)  
            ans[a[i].id] = 1;
        for(int i = 1; i <= n; ++i)
            printf("%d", ans[i]);
        printf("\n");
    }
}
```
