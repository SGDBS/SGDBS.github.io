#### 代码

```
#include <cstdio>
#include <iostream>
#include <cstring>
#include <cmath>
using namespace std;
 
const int maxN = 1e5 + 7;
 
int T, n, a, b, ans[maxN];
 
 
int main()
{
    scanf("%d", &T);
    while(T--) {
        scanf("%d%d%d", &n, &a, &b);
        if(a + b + 2 > n || abs(a - b) > 1) {
            printf("-1\n");
            continue;
        }
        int l = 1, r = n, flag = (a > b) ? 1 : 0;
        for(int i = 1; i <= n; ++i) {
            if(flag)
                ans[i] = l++;
            else    
                ans[i] = r--;
            if(i <= a + b)
                flag ^= 1;    
        }
        for(int i = 1; i <= n; ++i)
            printf("%d ", ans[i]);
        printf("\n");  
    }
    return 0;
}
```
