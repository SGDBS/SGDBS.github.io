### 代码

```
#include <cstdio>
#include <iostream>
using namespace std;

const int maxN = 3e5 + 7;

int n, k;
long long b[maxN], cl[maxN], ans;

int main()
{
    scanf("%d%d", &n, &k);
    for(int i = 1; i <= n; ++i)
        scanf("%lld", &b[i]);
    long long sum = 0, cnt = 0;
    for(int i = n; i >= 1; --i) {
        sum -= cnt;
        cnt -= cl[i];
        b[i] -= sum;
        if(b[i] > 0) {
            int now = min(i, k);
            long long nd = b[i] / now + (b[i] % now == 0 ? 0 : 1);
            sum += nd * now;
            cnt += nd;
            ans += nd;
            if(i - now >= 1)
                cl[i - now] += nd;
        }
    }
    printf("%lld\n", ans);
    return 0;
}
```
