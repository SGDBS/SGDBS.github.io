### 代码

```
#include <cstdio>
#include <iostream>
#include <bitset>
using namespace std;

int T, n, a[105], b[105];

int main()
{
    scanf("%d", &T);
    while(T--) {
        int minSum = 0, maxSum = 0, rSum = 0;
        scanf("%d", &n);
        for(int i = 1; i <= n; ++i)
            scanf("%d", &a[i]);
        for(int i = 1; i <= n; ++i)
            scanf("%d", &b[i]);
        for(int i = 1; i <= n; ++i) {
            if(a[i] > b[i])
                swap(a[i], b[i]);
            minSum += a[i];
            maxSum += b[i];
            rSum += a[i] * a[i] + b[i] * b[i];
        }
        bitset<105 * 105> dp;
        dp[0] = 1;
        for(int i = 1; i <= n; ++i)
            dp |= dp << (b[i] - a[i]);
        int ans = maxSum * maxSum + minSum * minSum;
        for(int i = 0; i < maxSum - minSum; ++i)
            if(dp[i])
                ans = min(ans, (maxSum - i) * (maxSum - i) + (minSum + i) * (minSum + i));
        printf("%d\n", (n - 2) * rSum + ans);
        
    }
    return 0;
}
```
