### 代码

```
#include <cstdio>
#include <iostream>
#include <vector>
using namespace std;

const int maxN = 2e4 + 7;

int T, n, res[maxN];

int main()
{
    scanf("%d", &T);
    while(T--) {
        cin >> n;
        vector<int> imp;
        for(int i = 0; i < n; ++i) {
            cout << "? " << i + 1 << " " << (i+1) % n + 1 << " " << (i+2) % n + 1 << endl;
            cin >> res[i];
        }
        for(int i = 0; i < n; ++i) {
            if(res[i] != res[(i + 1) % n]) {
                if(res[i] == 0)
                    imp.push_back(i + 1);
                else
                    imp.push_back((i + 3) % n + 1);
                for(int j = 0; j < n; ++j) {
                    if(j != i && j != (i + 3) % n) {
                        cout << "? " << i + 1 << " " << (i+3) % n + 1 << " " << j + 1 << endl;
                        int pd;
                        cin >> pd;
                        if(pd == 0)
                            imp.push_back(j + 1);
                    }
                }
                break;
            }
        }
        cout << "! " << imp.size();
        for(int x : imp) cout << " " << x;
        cout << endl;
    }
    return 0;
}
```
