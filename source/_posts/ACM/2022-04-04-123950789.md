### 代码

```
#include <cstdio>
#include <iostream>
#include <algorithm>
using namespace std;

const int maxN = 3e5 + 7;

int n, a[maxN];

int main()
{
    scanf("%d", &n);
    for(int i = 1; i <= n; ++i)
        scanf("%d", &a[i]);
    sort(a + 1, a + 1 + n);
    if(a[n] - a[n - 1] > 1)
        printf("Alice\n");
    else {
        if((a[n] - (n - 1)) % 2)
            printf("Alice\n");
        else
            printf("Bob\n");
    }
    return 0;
}
```
