### 代码

```
#include <cstdio>
#include <iostream>
using namespace std;

const int maxN = 2e5 + 7;
int T, n;
long long a, b, x[maxN], sum1[maxN];

int main()
{
    scanf("%d", &T);
    while(T--) {
        scanf("%d%lld%lld", &n, &a, &b);
        for(int i = 1; i <= n; ++i) {
            scanf("%lld", &x[i]);
            sum1[i] = sum1[i - 1] + x[i];
        }
        long long ans = 1LL << 62;
        for(int i = 0; i <= n; ++i) {
            long long forward = x[i] * b, after = (sum1[n] - sum1[i] - (n - i) * x[i]) * b;
            ans = min(ans, forward + after + a * x[i]);
        }
        printf("%lld\n", ans);
    }
    return 0;
}
```
