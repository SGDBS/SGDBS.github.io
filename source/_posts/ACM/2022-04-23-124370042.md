### 代码

```
#include <cstdio>
#include <iostream>
using namespace std;

int n;
const int maxN = 105;

double p[maxN], ans;

int main()
{
    scanf("%d", &n);
    for(int i = 1; i <= n; ++i)
        scanf("%lf", &p[i]);
    for(int i = 1; i <= n; ++i) 
        for(int j = 1; j <= n; ++j)
            if(i != j)
                ans += p[i] * p[j] / (p[i] + p[j]);
    printf("%lf\n", ans);
    return 0;
}
```
