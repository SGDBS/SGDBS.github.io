### 代码

```
#include <cstdio>
#include <iostream>
#include <cmath>
using namespace std;

long long c, h1, h2, w1, w2;

int main()
{
	cin >> c >> h1 >> h2 >> w1 >> w2;
	if(w1 > w2) 
        swap(h1, h2), swap(w1, w2);
	if(w2 > sqrt(c))
	{
		long long ans = 0;
		for(int i = 0; i * w2 <= c; i++) 
            ans = max(ans, i * h2 + (c - i * w2) / w1 * h1);
		cout << ans;
	}
	else
	{
		long long ans = 0;
		if(h1 * w2 < w1 * h2) 
            swap(h1, h2), swap(w1, w2);
		for(int i = 0; i <= w1; i++) ans = max(ans, i * h2 + (c - i * w2) / w1 * h1);
		cout << ans;
	}
    return 0;
}
```
